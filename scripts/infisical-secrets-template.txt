# Infisical Secrets — Migration Template
# ========================================
# After logging into Infisical and creating a project, add these secrets
# to the "dev" environment. For prod, add them to the "prod" environment
# with approval-gated access.
#
# Commands to run after `infisical login`:
#
#   infisical secrets set HERMES_CUSTOM_API_AIAND_COM_API_KEY="sk-..." --env=dev
#   infisical secrets set SUDO_PASSWORD="..." --env=dev
#   infisical secrets set EXA_API_KEY="..." --env=dev
#
# For prod (approval-gated):
#   infisical secrets set HERMES_CUSTOM_API_AIAND_COM_API_KEY="sk-..." --env=prod
#   infisical secrets set SUDO_PASSWORD="..." --env=prod
#   infisical secrets set EXA_API_KEY="..." --env=prod
#
# Then configure approval workflows in the Infisical dashboard:
#   https://app.infisical.com
#   Settings → Access Controls → Access Requests → enable for prod env

[dev]
HERMES_CUSTOM_API_AIAND_COM_API_KEY = "your-api-key-here"
SUDO_PASSWORD = "your-sudo-password-here"
EXA_API_KEY = "your-exa-key-here"

[prod]
# Same keys, but approval-gated. Agents must request access,
# you approve, and the lease auto-expires.
HERMES_CUSTOM_API_AIAND_COM_API_KEY = "your-prod-api-key-here"
SUDO_PASSWORD = "your-prod-sudo-password-here"
EXA_API_KEY = "your-prod-exa-key-here"
